#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import os
import sys
import random
import subprocess
import shutil
from pathlib import Path


def get_random_cover_image(cover_dir="~/tcardgen/cover"):
    """
    从封面目录中随机选择一个图片文件
    """
    # 扩展用户目录的 ~
    cover_path = os.path.expanduser(cover_dir)

    if not os.path.exists(cover_path):
        print(f"错误: 封面目录不存在: {cover_path}")
        return None

    # 获取所有图片文件
    image_extensions = {".png", ".jpg", ".jpeg", ".webp", ".gif"}
    image_files = []

    for file in os.listdir(cover_path):
        if os.path.isfile(os.path.join(cover_path, file)):
            ext = os.path.splitext(file)[1].lower()
            if ext in image_extensions:
                image_files.append(file)

    if not image_files:
        print(f"错误: 在目录 {cover_path} 中没有找到图片文件")
        return None

    # 随机选择一个图片
    selected_image = random.choice(image_files)
    return selected_image


def run_hugo_new_post(cardname, blog_dir="/Users/mac/study/HyBlog"):
    """
    执行 hugo new post 命令创建新文章
    """
    try:
        # 构建命令
        cmd = [
            "hugo",
            "new",
            f"post/{cardname}/{cardname}.md"
        ]
        
        print(f"正在创建新文章: {cardname}")
        print(f"执行命令: {' '.join(cmd)}")
        
        # 切换到博客目录执行命令
        result = subprocess.run(cmd, cwd=blog_dir, capture_output=True, text=True, timeout=30)

        if result.returncode == 0:
            print(f"✓ 文章创建成功: {cardname}")
            return True
        else:
            print(f"✗ 文章创建失败:")
            print(f"错误输出: {result.stderr}")
            return False

    except subprocess.TimeoutExpired:
        print(f"✗ 命令执行超时")
        return False
    except FileNotFoundError:
        print("✗ 找不到 hugo 命令，请确保已安装 Hugo")
        return False
    except Exception as e:
        print(f"✗ 执行命令时发生错误: {e}")
        return False


def generate_cover(cardname, cover_image, config_dir="/Users/mac/tcardgen", blog_dir="/Users/mac/study/HyBlog"):
    """
    使用 tcardgen 生成封面
    """
    try:
        # 构建路径
        config_path = os.path.join(config_dir, "config.yaml")
        font_dir = os.path.expanduser("~/tcardgen/font")
        cover_template = os.path.expanduser(f"~/tcardgen/cover/{cover_image}")
        output_path = os.path.join(blog_dir, "static", "img", f"{cardname}_cover.png")
        md_file_path = os.path.join(blog_dir, "content", "post", cardname, "index.md")
        
        # 确保输出目录存在
        os.makedirs(os.path.dirname(output_path), exist_ok=True)
        
        # 构建命令
        cmd = [
            "tcardgen",
            "-c", config_path,
            "-f", font_dir,
            "-o", output_path,
            "-t", cover_template,
            md_file_path
        ]
        
        print(f"正在生成封面: {cardname}")
        print(f"使用模板: {cover_template}")
        print(f"输出到: {output_path}")
        print(f"执行命令: {' '.join(cmd)}")
        
        # 执行命令
        result = subprocess.run(cmd, capture_output=True, text=True, timeout=60)

        if result.returncode == 0:
            print(f"✓ 封面生成成功: {output_path}")
            return True
        else:
            print(f"✗ 封面生成失败:")
            print(f"错误输出: {result.stderr}")
            return False

    except subprocess.TimeoutExpired:
        print(f"✗ 命令执行超时")
        return False
    except FileNotFoundError:
        print("✗ 找不到 tcardgen 命令，请确保已安装 tcardgen")
        return False
    except Exception as e:
        print(f"✗ 执行命令时发生错误: {e}")
        return False


def rename_md_file(cardname, blog_dir="/Users/mac/study/HyBlog"):
    """
    重命名 .md 文件为 index.md
    """
    try:
        original_path = os.path.join(blog_dir, "content", "post", cardname, f"{cardname}.md")
        new_path = os.path.join(blog_dir, "content", "post", cardname, "index.md")
        
        # 检查原文件是否存在
        if not os.path.exists(original_path):
            print(f"✗ 原文件不存在: {original_path}")
            return False
        
        # 重命名文件
        shutil.move(original_path, new_path)
        print(f"✓ 文件重命名成功: {original_path} -> {new_path}")
        return True
        
    except Exception as e:
        print(f"✗ 重命名文件时发生错误: {e}")
        return False


def main():
    """
    主函数
    """
    print("=" * 50)
    print("Hugo 文章创建与封面生成工具")
    print("=" * 50)
    
    # 检查参数
    if len(sys.argv) != 2:
        print("使用方法: python posthugo.py <cardname>")
        print("示例: python posthugo.py my-new-post")
        sys.exit(1)
    
    cardname = sys.argv[1]
    
    # 检查 cardname 是否合法
    if not cardname or not cardname.isalnum() and '-' not in cardname and '_' not in cardname:
        print("错误: cardname 只能包含字母、数字、连字符和下划线")
        sys.exit(1)
    
    print(f"处理文章: {cardname}")
    
    # 1. 创建新文章
    if not run_hugo_new_post(cardname):
        print("✗ 文章创建失败，退出")
        sys.exit(1)
    
    # 2. 重命名文件
    if not rename_md_file(cardname):
        print("✗ 文件重命名失败，退出")
        sys.exit(1)
    
    # 3. 随机选择封面图片
    print("\n正在选择随机封面...")
    cover_image = get_random_cover_image()
    
    if not cover_image:
        print("✗ 无法获取封面图片，退出")
        sys.exit(1)
    
    print(f"选择的封面模板: {cover_image}")
    
    # 4. 生成封面
    if not generate_cover(cardname, cover_image):
        print("✗ 封面生成失败")
        # 不退出，因为文章已经创建成功
    
    print("\n" + "=" * 50)
    print(f"处理完成!")
    print(f"文章位置: /Users/mac/study/HyBlog/content/post/{cardname}/index.md")
    print(f"封面位置: /Users/mac/study/HyBlog/static/img/{cardname}_cover.png")
    print("=" * 50)


if __name__ == "__main__":
    main()

会从~/tcardgen/cover中选择一个动漫图片,然后使用tcardgen来生成封面,最后调用相应的hugo new post/文章.md 来生成新的文章,将照片放入./static/cover/目录中.
